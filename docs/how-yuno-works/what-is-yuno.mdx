```javascript Node.js
const res = await fetch("https://api-sandbox.y.uno/v1/payments", {
  method: "POST",
  headers: {
    "public-api-key": process.env.YUNO_PUBLIC_API_KEY,
    "private-secret-key": process.env.YUNO_PRIVATE_SECRET_KEY,
    "X-Idempotency-Key": crypto.randomUUID(),
    "content-type": "application/json",
  },
  body: JSON.stringify({
    account_id: process.env.YUNO_ACCOUNT_ID,
    merchant_order_id: "order-0001",

    amount: { currency: "USD", value: 25 },

  }),
});
```

```python Python
import os
import uuid
import requests

res = requests.post(
    "https://api-sandbox.y.uno/v1/payments",
    headers={
        "public-api-key": os.environ["YUNO_PUBLIC_API_KEY"],
        "private-secret-key": os.environ["YUNO_PRIVATE_SECRET_KEY"],
        "X-Idempotency-Key": str(uuid.uuid4()),
    },
    json={
        "account_id": os.environ["YUNO_ACCOUNT_ID"],
        "merchant_order_id": "order-0001",
        "description": "First test payment",
        "country": "US",
        "amount": {"currency": "USD", "value": 25},
        "workflow": "DIRECT",
        "payment_method": {
            "type": "CARD",
            "detail": {"card": {"card_data": {
                "number": "4507990000000002",
                "expiration_month": 11,
                "expiration_year": 28,
                "security_code": "123",
                "holder_name": "John Doe",
            }}},
        },
    },
)
print(res.json()["status"])  # SUCCEEDED
```

<section className="yd-section" id="objects">
  <div className="yd-eyebrow-sm">02 / Core objects</div>
  <h2 className="yd-h2">Four objects, in the order you use them.</h2>
  <p className="yd-sub">Every payment follows this sequence, with any method and any provider.</p>
  <ol className="yd-objects">
    <li>
      <a className="yd-object" href="/docs/basic-concepts/customers">
        <span className="yd-object-num">1</span>
        <span className="yd-object-title">Customer <span className="yd-object-opt">Optional</span></span>
        <code>POST /v1/customers</code>
        <span className="yd-object-body">The payer. Stores their details and saved payment methods for reuse.</span>
      </a>
    </li>
    <li>
      <a className="yd-object" href="/docs/basic-concepts/sessions">
        <span className="yd-object-num">2</span>
        <span className="yd-object-title">Checkout session</span>
        <code>POST /v1/checkout/sessions</code>
        <span className="yd-object-body">The amount and country. The SDK uses it to show the payment methods your account has enabled.</span>
      </a>
    </li>
    <li>
      <a className="yd-object" href="/docs/basic-concepts/payments-1">
        <span className="yd-object-num">3</span>
        <span className="yd-object-title">Payment</span>
        <code>POST /v1/payments</code>
        <span className="yd-object-body">The charge. It holds the transactions: authorization, capture, and refund.</span>
      </a>
    </li>
    <li>
      <a className="yd-object" href="/docs/basic-concepts/webhooks-1">
        <span className="yd-object-num">4</span>
        <span className="yd-object-title">Webhook</span>
        <code>Yuno → your server</code>
        <span className="yd-object-body">Yuno sends each status change to a URL you register in the Dashboard or by API.</span>
      </a>
    </li>
  </ol>
  <p className="yd-foot">The SDK turns card data into a one-time token (OTT) that you send with the payment. The Direct API skips the session and sends card data from your server, which requires PCI DSS. Full sequence: <a href="/docs/how-yuno-works/how-yuno-payment-flow-works">Payment flow</a>.</p>
