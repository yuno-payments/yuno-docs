```bash cURL
  -d '{
    "type": "payment",

```

```javascript Node.js
app.post("/webhooks/yuno", express.raw({ type: "application/json" }), (req, res) => {
  // 1. Verify the HMAC signature before trusting the body
  //    (see Verify webhook signatures)
  if (!verifyYunoSignature(req)) return res.sendStatus(401);
  const event = JSON.parse(req.body);

  // 2. Yuno retries on failure: skip a payment you already handled
  if (alreadyProcessed(event.data.payment.id)) return res.sendStatus(200);

  if (event.type_event === "payment.purchase") {
    updateOrder(event.data.payment);
  }

  // Respond fast
  res.sendStatus(200);
});
```

```python Python
@app.post("/webhooks/yuno")
def yuno_webhook():
    # 1. Verify the HMAC signature before trusting the body
    #    (see Verify webhook signatures)
    if not verify_yuno_signature(request):
        return "", 401
    event = request.get_json()

    # 2. Yuno retries on failure: skip a payment you already handled
    if already_processed(event["data"]["payment"]["id"]):
        return "", 200

    if event["type_event"] == "payment.purchase":
        update_order(event["data"]["payment"])

    # Respond fast
    return "", 200
```
