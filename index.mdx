```javascript
// 1. Create checkout session (server-side)
const session = await fetch("https://api-sandbox.y.uno/v1/checkout/sessions", {
  method: "POST",
  headers: {
    "public-api-key": process.env.YUNO_PUBLIC_KEY,
    "private-secret-key": process.env.YUNO_SECRET_KEY,
    "Content-Type": "application/json"
  },
  body: JSON.stringify({
    account_id: process.env.YUNO_ACCOUNT_ID,
    merchant_order_id: "order-123",
    payment_description: "Test payment",
    country: "US",
    amount: { currency: "USD", value: 10 }
  })
}).then(r => r.json());

// 2. Create payment with SDK token (in yunoCreatePayment callback)
const payment = await fetch("https://api-sandbox.y.uno/v1/payments", {
  method: "POST",
  headers: {
    "public-api-key": process.env.YUNO_PUBLIC_KEY,
    "private-secret-key": process.env.YUNO_SECRET_KEY,
    "X-Idempotency-Key": crypto.randomUUID(),
    "Content-Type": "application/json"
  },
  body: JSON.stringify({
    account_id: process.env.YUNO_ACCOUNT_ID,
    description: "Test payment",
    country: "US",
    merchant_order_id: "order-123",
    amount: { currency: "USD", value: 10 },
    checkout: { session: session.checkout_session },

  })
}).then(r => r.json());
// payment.status === "SUCCEEDED" → done
```

```bash
# 1. Create checkout session
curl -X POST https://api-sandbox.y.uno/v1/checkout/sessions \
  -H "public-api-key: $YUNO_PUBLIC_KEY" \
  -H "private-secret-key: $YUNO_SECRET_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "account_id": "'"$YUNO_ACCOUNT_ID"'",
    "merchant_order_id": "order-123",
    "payment_description": "Test payment",
    "country": "US",
    "amount": { "currency": "USD", "value": 10 }
  }'

# 2. Create payment (use checkout_session from step 1)
curl -X POST https://api-sandbox.y.uno/v1/payments \
  -H "public-api-key: $YUNO_PUBLIC_KEY" \
  -H "private-secret-key: $YUNO_SECRET_KEY" \
  -H "X-Idempotency-Key: $(uuidgen)" \
  -H "Content-Type: application/json" \
  -d '{
    "account_id": "'"$YUNO_ACCOUNT_ID"'",
    "description": "Test payment",
    "country": "US",
    "merchant_order_id": "order-123",
    "amount": { "currency": "USD", "value": 10 },
    "checkout": { "session": "<checkout_session>" },
    "payment_method": { "token": "<one_time_token>" }
  }'
```
